**Contexto (Nivel 1)**

```mermaid
C4Context
  title Sistema de entrega de provisiones — Contexto

  Enterprise_Boundary(b0, "Ecosistema de Entrega de Provisiones") {
    
    %% CAPA SUPERIOR
    Boundary(b1, "Actores (Usuarios)") {
      Person(admin, "Administrador", "Gestiona <br>cuentas y solicitudes")
      Person(solicitante, "Solicitante", "Envía y cancela <br>solicitudes")
      Person(remitente, "Remitente", "Envía y gestiona<br> paquetes")
    }

    %% CAPA CENTRAL
    Boundary(b2, "Core del Negocio") {
      System(sistema, "Sistema de entrega<br> de provisiones", "Coordina y controla <br>solicitudes y entregas")
    }

    %% CAPA INFERIOR
    Boundary(b3, "Sistemas Externos y Soporte") {
      System_Ext(apiClima, "API Clima", "Provee datos meteorológicos")
      System_Ext(apiMapa, "API Mapa", "Provee datos de coordenadas")
      System_Ext(notif, "Sistema de <br>notificaciones", "Envía notificaciones <br>a los usuarios")
    }
  }

  %% Relaciones de Capa 1 a Capa 2
  Rel(admin, sistema, "Administra<br> solicitudes y cuentas")
  Rel(solicitante, sistema, "Gestiona<br> pedidos")
  Rel(remitente, sistema, "Gestiona<br> envíos")

  %% Relaciones de Capa 2 a Capa 3
  Rel(sistema, apiClima, "Consulta <br>condiciones <br>del clima")
  Rel(sistema, apiMapa, "Consulta<br> coordenadas")
  Rel(sistema, notif, "Envía<br> notificaciones")

  %% Notificaciones de vuelta a usuarios (Capa 3 a Capa 1)
  Rel(notif, solicitante, "Notifica")
  Rel(notif, remitente, "Notifica")

    UpdateRelStyle(admin, sistema, $textColor="black", $lineColor="black", $offsetX="-50", $offsetY="-20")
    UpdateRelStyle(sistema, apiClima, $textColor="black", $lineColor="black", $offsetX="-50", $offsetY="20")
    UpdateRelStyle(sistema, notif, $textColor="black", $lineColor="black", $offsetX="-80", $offsetY="20")
    UpdateRelStyle(solicitante, sistema, $textColor="black", $lineColor="black", $offsetX="-70", $offsetY="-15")
    UpdateRelStyle(remitente, sistema, $textColor="black", $lineColor="black", $offsetX="-15", $offsetY="-15")
    UpdateLayoutConfig($c4ShapeInRow="3", $c4BoundaryInRow="3")
```

  Rel(stockSvc, intBdd, "Usa para leer y modificar")
  Rel(calcTray, intClima, "Usa")
  Rel(calcTray, intMapa, "Usa")
  Rel(intBdd, bdd, "Lee y escribe")
  Rel(intClima, apiClima, "Hace llamadas API a")
  Rel(intMapa, apiMapa, "Hace llamadas API a")
